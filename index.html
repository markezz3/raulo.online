function handleLogin(event) { event.preventDefault(); const username = document.getElementById("username").value.toLowerCase(); // Convertir a minúsculas para evitar problemas de mayúsculas const password = document.getElementById("password").value; // Credenciales válidas const validCredentials = [ { user: "admin", pass: "admin" }, { user: "guest", pass: "guest" } ]; // Verificar si las credenciales son válidas const isValid = validCredentials.some(cred => cred.user === username && cred.pass === password ); if (isValid) { // Guardar que el usuario está autenticado (opcional) localStorage.setItem("isLoggedIn", "true"); localStorage.setItem("username", username); // Redirigir a portal.html window.location.href = "portal.html"; } else { alert("❌ Usuario o contraseña incorrectos.\n\nIntenta:\n- Contactar a un administrador\n- Presiona aceptar e intenta de nuevo"); document.getElementById("loginForm").reset(); } } // Verificar si ya está logueado window.addEventListener("load", function() { const isLoggedIn = localStorage.getItem("isLoggedIn"); if (isLoggedIn === "true") { // Si ya estaba logueado, ocultar el login automáticamente // (Esto permite que recarguen la página sin volver a loguear) document.getElementById("loginOverlay").classList.add("hidden"); } });
